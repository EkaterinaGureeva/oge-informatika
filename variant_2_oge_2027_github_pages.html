Паскаль
var s, n, A: integer;
begin
  readln(s);
  readln(n);
  if (s > 20) or (n > A)
    then writeln('ДА')
    else writeln('НЕТ')
end.
Python
s = int(input())
n = int(input())
if s > 20 or n > A:
    print("ДА")
else:
    print("НЕТ")
C++
#include <iostream>
using namespace std;
int main() {
 int s, n;
 cin >> s >> n;
 if (s > 20 || n > A)
   cout << "ДА";
 else
   cout << "НЕТ";
 return 0;
}
