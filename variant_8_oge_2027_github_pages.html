Задание 5Часть 1

  readln(t);

    else writeln('НЕТ')
end.
Python
s = int(input())
t = int(input())
A = int(input())
if s > A or t > 11:
    print('ДА')
else:
    print('НЕТ')
C++
#include <iostream>
using namespace std;
int main() {
  int s, t, A;
  cin >> s >> t;
  cin >> A;
  if (s > A || t > 11)
    cout << "ДА";
  else
    cout << "НЕТ";
  return 0;
}
