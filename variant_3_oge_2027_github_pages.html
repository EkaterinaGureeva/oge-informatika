s = int(input())
t = int(input())
if abs(s) < 3 or abs(t) > 3:

#include <iostream>
#include <cmath>
using namespace std;
int main() {
  int s, t;
  cin >> s >> t;
  if (abs(s) < 3 || abs(t) > 3)
    cout << "ДА";
  else cout << "НЕТ";
}
