Паскаль
var k, n: integer;
begin
  readln(k);
  readln(n);
  if not((k <= 25 and k > 73) or n = 47)
    then writeln('ДА')
    else writeln('НЕТ')
end.
Python
k = int(input())
n = int(input())
if not((k <= 25 and k > 73) or n == 47):
    print('ДА')
else:
    print('НЕТ')
C++
#include <iostream>
using namespace std;
int main() {
  int k, n;
  cin >> k >> n;
  if (!((k <= 25 && k > 73) || n == 47))
    cout << "ДА";
  else
    cout << "НЕТ";
  return 0;
}
